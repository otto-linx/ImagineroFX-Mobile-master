// Función para obtener el saludo según la hora
function getSaludo() {
  const hora = new Date().getHours();
  if (hora >= 6 && hora < 12) return 'Buenos días';
  if (hora >= 12 && hora < 19) return 'Buenas tardes';
  return 'Buenas noches';
}

ProductScreen.js

import React, { useContext, useState } from 'react';
import {
  View,
  Text,
  Image,
  StyleSheet,
  ScrollView,
  TouchableOpacity,
  SafeAreaView,
  Share,
  Alert,
  Modal,
  Dimensions
} from 'react-native';
import { Ionicons } from '@expo/vector-icons';
import { CartContext } from '../context/CartContext';

const SCREEN_WIDTH = Dimensions.get('window').width;

export default function ProductScreen({ route, navigation }) {
  const { product } = route.params;
  const { addToCart } = useContext(CartContext);
  const [quantity, setQuantity] = useState(1);
  const [showSuccessModal, setShowSuccessModal] = useState(false);
  const [isFavorite, setIsFavorite] = useState(false);
  const [expanded, setExpanded] = useState(false);

  if (!product) {
    return (
      <View style={styles.container}>
        <Text style={styles.errorText}>Producto no disponible.</Text>
      </View>
    );
  }

  const handleAddToCart = () => {
    addToCart({ ...product, quantity });
    setShowSuccessModal(true);
  };

  const increaseQuantity = () => {
    setQuantity(prev => Math.min(prev + 1, 10));
  };

  const decreaseQuantity = () => {
    setQuantity(prev => Math.max(prev - 1, 1));
  };

  const shareProduct = async () => {
    try {
      await Share.share({
        message: `¡Mira este increíble producto en ImagineroFX! ${product.name} - $${product.price}\n${product.description || ''}`,
        url: 'https://imaginero.fx',
        title: product.name
      });
    } catch (error) {
      Alert.alert('Error', 'No se pudo compartir el producto');
    }
  };

  const renderRatingStars = () => {
    const rating = product.rating || 4.5;
    const stars = [];
    for (let i = 1; i <= 5; i++) {
      stars.push(
        <Ionicons
          key={i}
          name={i <= Math.floor(rating) ? 'star' : 'star-outline'}
          size={20}
          color={i <= rating ? '#FFD700' : '#ccc'}
        />
      );
    }
    return <View style={styles.ratingStars}>{stars}</View>;
  };

  const goToCart = () => {
    setShowSuccessModal(false);
    navigation.navigate('Main', { screen: 'Cart' });
  };

  const continueShopping = () => {
    setShowSuccessModal(false);
  };

  const buyNow = () => {
    addToCart({ ...product, quantity });
    navigation.navigate('Checkout');
  };

  const toggleFavorite = () => {
    setIsFavorite(!isFavorite);
    Alert.alert(
      isFavorite ? 'Eliminado de favoritos' : 'Añadido a favoritos',
      isFavorite ? 'El producto ha sido eliminado de tu lista de deseados' : 'El producto ha sido añadido a tu lista de deseados'
    );
  };

  // Productos relacionados filtrados por categoría del producto actual
  const relatedProducts = products.filter(
    (item) =>
      item.category === product.category && item.id !== product.id
  );

  return (
    <SafeAreaView style={styles.safeArea}>
      <ScrollView style={styles.scrollView} showsVerticalScrollIndicator={false}>
        {/* Header */}
        <View style={styles.header}>
          <TouchableOpacity onPress={() => navigation.goBack()} style={styles.backButton}>
            <Ionicons name="arrow-back" size={28} color="#333" />
          </TouchableOpacity>
          {/* Botón de favoritos eliminado */}
        </View>

        {/* Imagen del producto */}
        <View style={styles.imageShadowContainer}>
          <Image
            source={product.image}
            style={styles.image}
            resizeMode="cover"
          />
          <View style={styles.badge}>
            <Text style={styles.badgeText}>Escena Cinematográfica</Text>
          </View>
        </View>

        {/* Información principal */}
        <View style={styles.infoContainer}>
          <Text style={styles.title}>{product.name}</Text>
          <View style={styles.priceRatingRow}>
            <Text style={styles.price}>${Number(product.price).toLocaleString('es-CO', {minimumFractionDigits: 0})}</Text>
            <View style={styles.ratingContainer}>
              {renderRatingStars()}
              <Text style={styles.ratingNumber}>{product.rating}</Text>
            </View>
          </View>
          {product.category && (
            <Text style={styles.category}>{product.category}</Text>
          )}
        </View>

        {/* Ficha técnica */}
        <View style={styles.techCard}>
          <Text style={styles.techTitle}>Ficha técnica</Text>
          <View style={styles.techRow}>
            <Ionicons name="cube-outline" size={18} color="#d32f2f" style={styles.techIcon} />
            <Text style={styles.techLabel}>Material:</Text>
            <Text style={styles.techValue}>{product.material || 'Madera y metal'}</Text>
          </View>
          <View style={styles.techRow}>
            <Ionicons name="resize-outline" size={18} color="#d32f2f" style={styles.techIcon} />
            <Text style={styles.techLabel}>Dimensiones:</Text>
            <Text style={styles.techValue}>{product.dimensions || 'Personalizables'}</Text>
          </View>
          <View style={styles.techRow}>
            <Ionicons name="color-palette-outline" size={18} color="#d32f2f" style={styles.techIcon} />
            <Text style={styles.techLabel}>Colores:</Text>
            <Text style={styles.techValue}>{product.colors || 'Personalizables'}</Text>
          </View>
        </View>

        {/* Descripción */}
        {product.description && (
          <View style={styles.sectionCard}>
            <Text style={styles.sectionTitle}>Descripción</Text>
            <Text style={styles.description}>{product.description}</Text>
          </View>
        )}

        {/* Características destacadas */}
        <View style={styles.sectionCard}>
          <Text style={styles.sectionTitle}>Características</Text>
          <View style={styles.featureList}>
            <View style={styles.featureItem}>
              <Ionicons name="checkmark-circle" size={20} color="#4CAF50" />
              <Text style={styles.featureText}>Fabricación personalizada</Text>
            </View>
            <View style={styles.featureItem}>
              <Ionicons name="checkmark-circle" size={20} color="#4CAF50" />
              <Text style={styles.featureText}>Entrega en todo el país</Text>
            </View>
            <View style={styles.featureItem}>
              <Ionicons name="checkmark-circle" size={20} color="#4CAF50" />
              <Text style={styles.featureText}>Garantía de calidad cinematográfica</Text>
            </View>
          </View>
        </View>

        {/* Selector de cantidad */}
        <View style={styles.quantityContainer}>
          <Text style={styles.quantityLabel}>Cantidad</Text>
          <View style={styles.quantitySelector}>
            <TouchableOpacity
              onPress={decreaseQuantity}
              style={styles.quantityButton}
              activeOpacity={0.7}
            >
              <Ionicons name="remove" size={20} color="#fff" />
            </TouchableOpacity>
            <Text style={styles.quantityText}>{quantity}</Text>
            <TouchableOpacity
              onPress={increaseQuantity}
              style={styles.quantityButton}
              activeOpacity={0.7}
            >
              <Ionicons name="add" size={20} color="#fff" />
            </TouchableOpacity>
          </View>
        </View>

        {/* Botones de acción */}
        <View style={styles.buttonContainer}>
          <TouchableOpacity
            style={styles.cartButton}
            onPress={handleAddToCart}
            activeOpacity={0.8}
          >
            <Ionicons name="cart" size={24} color="#fff" />
            <Text style={styles.cartButtonText}>Agregar al carrito</Text>
          </TouchableOpacity>
          <TouchableOpacity
            style={styles.buyButton}
            onPress={buyNow}
            activeOpacity={0.8}
          >
            <Ionicons name="flash" size={24} color="#fff" />
            <Text style={styles.buyButtonText}>Comprar ahora</Text>
          </TouchableOpacity>
          {/* Botón de compartir eliminado */}
        </View>

        {/* Recomendaciones */}
        <View style={styles.recommendationsContainer}>
          <TouchableOpacity
            onPress={() => setExpanded((prev) => !prev)}
            activeOpacity={0.8}
            style={[
              styles.recommendationsHeader,
              expanded && styles.recommendationsHeaderExpanded,
            ]}
          >
            <Text style={styles.recommendationsTitle}>
              Productos relacionados
            </Text>
            <Ionicons
              name={expanded ? 'chevron-up' : 'chevron-down'}
              size={22}
              color="#d32f2f"
            />
          </TouchableOpacity>
          {expanded && (
            <ScrollView
              horizontal
              showsHorizontalScrollIndicator={false}
              style={styles.recommendationsList}
            >
              {relatedProducts.length === 0 ? (
                <Text style={{ color: '#888', marginLeft: 10 }}>
                  No hay más productos en esta categoría.
                </Text>
              ) : (
                relatedProducts.map((item) => (
                  <TouchableOpacity
                    key={item.id}
                    style={styles.recommendationCard}
                    onPress={() => navigation.push('Product', { product: item })}
                  >
                    <Image
                      source={item.image}
                      style={styles.recommendationImage}
                      resizeMode="cover"
                    />
                    <Text numberOfLines={1} style={styles.recommendationName}>
                      {item.name}
                    </Text>
                    <Text style={styles.recommendationPrice}>
                      ${item.price}
                    </Text>
                  </TouchableOpacity>
                ))
              )}
            </ScrollView>
          )}
        </View>
      </ScrollView>

      {/* Modal de éxito */}
      <Modal
        animationType="fade"
        transparent={true}
        visible={showSuccessModal}
        onRequestClose={() => setShowSuccessModal(false)}
      >
        <View style={styles.modalContainer}>
          <View style={styles.modalContent}>
            <View style={styles.successIcon}>
              <Ionicons name="checkmark-circle" size={60} color="#4CAF50" />
            </View>
            <Text style={styles.modalTitle}>¡Añadido al carrito!</Text>
            <Text style={styles.modalText}>
              {quantity} {quantity > 1 ? 'unidades' : 'unidad'} de {product.name}
            </Text>
            <View style={styles.modalButtons}>
              <TouchableOpacity
                style={styles.modalContinueButton}
                onPress={continueShopping}
              >
                <Text style={styles.modalContinueText}>Seguir comprando</Text>
              </TouchableOpacity>
              <TouchableOpacity
                style={styles.modalCartButton}
                onPress={goToCart}
              >
                <Text style={styles.modalCartText}>Ver carrito</Text>
              </TouchableOpacity>
            </View>
          </View>
        </View>
      </Modal>
    </SafeAreaView>
  );
}

import React, { useState } from 'react';
import {
  View,
  Text,
  TextInput,
  TouchableOpacity,
  StyleSheet,
  ScrollView,
  ActivityIndicator,
  KeyboardAvoidingView,
  Platform
} from 'react-native';
import { Ionicons } from '@expo/vector-icons';
import * as Yup from 'yup';
import { Formik } from 'formik';
import Toast from 'react-native-toast-message';
import { supabase } from '../supabaseClient';

// Validación de formulario con Yup
const validationSchema = Yup.object().shape({
  fullName: Yup.string().required('El nombre es requerido'),
  email: Yup.string()
    .email('Correo inválido')
    .required('El correo es requerido'),
  phone: Yup.string()
    .matches(/^[0-9]{10}$/, 'Debe ser un número válido')
    .required('Teléfono requerido'),
  password: Yup.string()
    .min(8, 'Mínimo 8 caracteres')
    .matches(/[A-Z]/, 'Debe contener una mayúscula')
    .matches(/[a-z]/, 'Debe contener una minúscula')
    .matches(/[0-9]/, 'Debe contener un número')
    .matches(/[^A-Za-z0-9]/, 'Debe contener un carácter especial')
    .required('Contraseña requerida'),
  confirmPassword: Yup.string()
    .oneOf([Yup.ref('password'), null], 'Las contraseñas no coinciden')
    .required('Confirma tu contraseña'),
});

export default function RegisterScreen({ navigation }) {
  const [passwordStrength, setPasswordStrength] = useState(0);
  const [showPassword, setShowPassword] = useState(false);
  const [showConfirm, setShowConfirm] = useState(false);
  const [loading, setLoading] = useState(false);

  // Evaluar fortaleza de contraseña
  const calculatePasswordStrength = (password) => {
    let score = 0;
    if (password.length >= 8) score++;
    if (/[A-Z]/.test(password)) score++;
    if (/[a-z]/.test(password)) score++;
    if (/[0-9]/.test(password)) score++;
    if (/[^A-Za-z0-9]/.test(password)) score++;
    setPasswordStrength(score);
  };

  const getStrengthColor = () => {
    if (passwordStrength <= 2) return '#d32f2f'; // Rojo
    if (passwordStrength === 3) return '#ffa726'; // Naranja
    return '#66bb6a'; // Verde
  };

  const handleRegister = async (values) => {
    setLoading(true);
    try {
      // 1. Registrar usuario
      const { data, error: signUpError } = await supabase.auth.signUp({
        email: values.email,
        password: values.password,
      });
      
      if (signUpError) throw signUpError;
      
      const userId = data.user.id;
      
      // 2. Guardar datos adicionales en la tabla de usuarios
      const { error: insertError } = await supabase
        .from('users')
        .insert([
          {
            id: userId,
            email: values.email,
            full_name: values.fullName,
            phone: values.phone,
            created_at: new Date().toISOString(),
          },
        ]);
      
      if (insertError) throw insertError;
      
      // 3. Iniciar sesión automáticamente
      const { error: signInError } = await supabase.auth.signInWithPassword({
        email: values.email,
        password: values.password,
      });
      
      if (signInError) throw signInError;
      
      // 4. Mostrar mensaje de éxito y redirigir
      Toast.show({
        type: 'success',
        text1: 'Registro exitoso',
        text2: 'Bienvenido a ImagineroFX',
        visibilityTime: 3000,
        onHide: () => {
          navigation.reset({
            index: 0,
            routes: [{ name: 'Main', params: { screen: 'Home' } }],
          });
        },
      });
    } catch (error) {
      Toast.show({
        type: 'error',
        text1: 'Error al registrarse',
        text2: error.message || 'Inténtalo más tarde',
      });
    } finally {
      setLoading(false);
    }
  };

  return (
    <KeyboardAvoidingView
      behavior={Platform.OS === 'ios' ? 'padding' : 'height'}
      style={styles.container}
    >
      <ScrollView contentContainerStyle={styles.scrollContainer}>
        <Text style={styles.title}>Crea tu cuenta</Text>
        <Text style={styles.subtitle}>Ingresa tus datos para registrarte</Text>

        <Formik
          initialValues={{
            fullName: '',
            email: '',
            phone: '',
            password: '',
            confirmPassword: '',
          }}
          validationSchema={validationSchema}
          onSubmit={handleRegister}
        >
          {({
            handleChange,
            handleBlur,
            handleSubmit,
            values,
            errors,
            touched,
          }) => (
            <>
              {/* Nombre completo */}
              <View style={styles.inputGroup}>
                <Ionicons name="person-outline" size={20} color="#999" style={styles.icon} />
                <TextInput
                  placeholder="Nombre completo *"
                  placeholderTextColor="#999"
                  value={values.fullName}
                  onChangeText={handleChange('fullName')}
                  onBlur={handleBlur('fullName')}
                  style={styles.input}
                  autoCapitalize="words"
                  autoCorrect={false}
                  keyboardType="default"
                />
              </View>
              {touched.fullName && errors.fullName && (
                <Text style={styles.errorText}>{errors.fullName}</Text>
              )}

              {/* Correo electrónico */}
              <View style={styles.inputGroup}>
                <Ionicons name="mail-outline" size={20} color="#999" style={styles.icon} />
                <TextInput
                  placeholder="Correo electrónico *"
                  placeholderTextColor="#999"
                  value={values.email}
                  onChangeText={handleChange('email')}
                  onBlur={handleBlur('email')}
                  style={styles.input}
                  keyboardType="email-address"
                  autoCapitalize="none"
                />
              </View>
              {touched.email && errors.email && (
                <Text style={styles.errorText}>{errors.email}</Text>
              )}

              {/* Teléfono */}
              <View style={styles.inputGroup}>
                <Ionicons name="call-outline" size={20} color="#999" style={styles.icon} />
                <TextInput
                  placeholder="Teléfono (ej: 3137978572) *"
                  placeholderTextColor="#999"
                  value={values.phone}
                  onChangeText={handleChange('phone')}
                  onBlur={handleBlur('phone')}
                  style={styles.input}
                  keyboardType="phone-pad"
                />
              </View>
              {touched.phone && errors.phone && (
                <Text style={styles.errorText}>{errors.phone}</Text>
              )}

              {/* Contraseña */}
              <View style={styles.inputGroup}>
                <Ionicons name="lock-closed-outline" size={20} color="#999" style={styles.icon} />
                <TextInput
                  placeholder="Contraseña *"
                  placeholderTextColor="#999"
                  value={values.password}
                  onChangeText={(text) => {
                    handleChange('password')(text);
                    calculatePasswordStrength(text);
                  }}
                  onBlur={handleBlur('password')}
                  style={styles.input}
                  secureTextEntry={!showPassword}
                />
                <TouchableOpacity
                  onPress={() => setShowPassword(!showPassword)}
                  style={styles.eyeIcon}
                  accessibilityLabel={showPassword ? "Ocultar contraseña" : "Mostrar contraseña"}
                >
                  <Ionicons
                    name={showPassword ? 'eye-off-outline' : 'eye-outline'}
                    size={20}
                    color="#999"
                  />
                </TouchableOpacity>
              </View>
              {touched.password && errors.password && (
                <Text style={styles.errorText}>{errors.password}</Text>
              )}

              {/* Barra de fortaleza de contraseña */}
              <View style={styles.strengthContainer}>
                <View style={styles.strengthBarContainer}>
                  <View
                    style={[
                      styles.strengthBar,
                      {
                        width: `${(passwordStrength / 5) * 100}%`,
                        backgroundColor: getStrengthColor(),
                      },
                    ]}
                  />
                </View>
                <Text style={styles.strengthLabel}>
                  {passwordStrength <= 2 && 'Contraseña débil'}
                  {passwordStrength === 3 && 'Contraseña media'}
                  {passwordStrength >= 4 && 'Contraseña fuerte'}
                </Text>
              </View>

              {/* Confirmar contraseña */}
              <View style={styles.inputGroup}>
                <Ionicons name="lock-closed-outline" size={20} color="#999" style={styles.icon} />
                <TextInput
                  placeholder="Confirmar contraseña *"
                  placeholderTextColor="#999"
                  value={values.confirmPassword}
                  onChangeText={handleChange('confirmPassword')}
                  onBlur={handleBlur('confirmPassword')}
                  style={styles.input}
                  secureTextEntry={!showConfirm}
                />
                <TouchableOpacity
                  onPress={() => setShowConfirm(!showConfirm)}
                  style={styles.eyeIcon}
                  accessibilityLabel={showConfirm ? "Ocultar confirmación" : "Mostrar confirmación"}
                >
                  <Ionicons
                    name={showConfirm ? 'eye-off-outline' : 'eye-outline'}
                    size={20}
                    color="#999"
                  />
                </TouchableOpacity>
              </View>
              {touched.confirmPassword && errors.confirmPassword && (
                <Text style={styles.errorText}>{errors.confirmPassword}</Text>
              )}

              {/* Botón de registro */}
              <TouchableOpacity
                style={styles.registerButton}
                onPress={handleSubmit}
                disabled={loading}
                activeOpacity={0.8}
                accessibilityLabel="Registrarse"
              >
                {loading ? (
                  <ActivityIndicator size="small" color="#fff" />
                ) : (
                  <Text style={styles.registerButtonText}>Registrarse</Text>
                )}
              </TouchableOpacity>

              {/* Requisitos de contraseña */}
              <View style={styles.requirementsContainer}>
                <Text style={styles.requirementText}>La contraseña debe contener:</Text>
                <View style={styles.requirementRow}>
                  <Ionicons 
                    name={values.password.length >= 8 ? "checkmark-circle" : "ellipse-outline"} 
                    size={16} 
                    color={values.password.length >= 8 ? "#4CAF50" : "#999"} 
                    style={styles.requirementIcon} 
                  />
                  <Text style={styles.requirementItem}>Mínimo 8 caracteres</Text>
                </View>
                <View style={styles.requirementRow}>
                  <Ionicons 
                    name={/[A-Z]/.test(values.password) ? "checkmark-circle" : "ellipse-outline"} 
                    size={16} 
                    color={/[A-Z]/.test(values.password) ? "#4CAF50" : "#999"} 
                    style={styles.requirementIcon} 
                  />
                  <Text style={styles.requirementItem}>Una letra mayúscula</Text>
                </View>
                <View style={styles.requirementRow}>
                  <Ionicons 
                    name={/[a-z]/.test(values.password) ? "checkmark-circle" : "ellipse-outline"} 
                    size={16} 
                    color={/[a-z]/.test(values.password) ? "#4CAF50" : "#999"} 
                    style={styles.requirementIcon} 
                  />
                  <Text style={styles.requirementItem}>Una letra minúscula</Text>
                </View>
                <View style={styles.requirementRow}>
                  <Ionicons 
                    name={/[0-9]/.test(values.password) ? "checkmark-circle" : "ellipse-outline"} 
                    size={16} 
                    color={/[0-9]/.test(values.password) ? "#4CAF50" : "#999"} 
                    style={styles.requirementIcon} 
                  />
                  <Text style={styles.requirementItem}>Un número</Text>
                </View>
                <View style={styles.requirementRow}>
                  <Ionicons 
                    name={/[^A-Za-z0-9]/.test(values.password) ? "checkmark-circle" : "ellipse-outline"} 
                    size={16} 
                    color={/[^A-Za-z0-9]/.test(values.password) ? "#4CAF50" : "#999"} 
                    style={styles.requirementIcon} 
                  />
                  <Text style={styles.requirementItem}>Un carácter especial</Text>
                </View>
              </View>

              {/* Enlace a Login */}
              <TouchableOpacity
                style={styles.loginContainer}
                onPress={() => navigation.navigate('Login')}
                accessibilityRole="button"
                accessibilityLabel="¿Ya tienes cuenta? Inicia sesión"
              >
                <Text style={styles.loginText}>¿Ya tienes cuenta? </Text>
                <Text style={styles.highlight}>Inicia sesión</Text>
              </TouchableOpacity>
            </>
          )}
        </Formik>
      </ScrollView>
      <Toast />
    </KeyboardAvoidingView>
  );
}

function getSaludo() {
  const hora = new Date().getHours();
  if (hora >= 6 && hora < 12) return 'Buenos días';
  if (hora >= 12 && hora < 19) return 'Buenas tardes';
  return 'Buenas noches';
}

// Navegación de pestañas
function MainTabs() {
  return (
    <Tab.Navigator
      screenOptions={({ route }) => ({
        headerShown: false,
        tabBarActiveTintColor: '#d32f2f',
        tabBarInactiveTintColor: 'gray',
        tabBarStyle: {
          backgroundColor: '#fff',
          borderTopWidth: 0,
          paddingBottom: 5,
          paddingTop: 5,
          height: 60,
        },
        tabBarIcon: ({ color, size, focused }) => {
          let iconName;
          switch (route.name) {
            case 'Home':
              iconName = focused ? 'home' : 'home-outline';
              break;
            case 'Cart':
              iconName = focused ? 'cart' : 'cart-outline';
              break;
            case 'Account':
              iconName = focused ? 'person' : 'person-outline';
              break;
            case 'Offers':
              iconName = focused ? 'pricetag' : 'pricetag-outline';
              break;
            default:
              iconName = 'ellipse-outline';
          }
          return <Ionicons name={iconName} size={size} color={color} />;
        },
      })}
    >
      <Tab.Screen name="Home" component={HomeScreen} options={{ title: 'Inicio' }} />
      <Tab.Screen name="Offers" component={SpecialOffersScreen} options={{ title: 'Ofertas' }} />
      <Tab.Screen name="Cart" component={CartScreen} options={{ title: 'Carrito' }} />
      <Tab.Screen name="Account" component={AccountScreen} options={{ title: 'Cuenta' }} />
    </Tab.Navigator>
  );
}

// Navegador principal
export default function AppNavigator() {
  const [isSessionChecked, setIsSessionChecked] = useState(false);
  const [isLoggedIn, setIsLoggedIn] = useState(false);

  useEffect(() => {
    const getCurrentSession = async () => {
      try {
        const { data: { session } } = await supabase.auth.getSession();
        setIsLoggedIn(!!session);
      } catch (error) {
        console.error("Error obteniendo sesión:", error);
      } finally {
        setIsSessionChecked(true);
      }
    };

    // Listener para cambios de autenticación
    const { data: { subscription } } = supabase.auth.onAuthStateChange(
      (event, session) => {
        setIsLoggedIn(!!session);
      }
    );

    getCurrentSession();

    return () => {
      subscription.unsubscribe();
    };
  }, []);

  if (!isSessionChecked) {
    return <LoadingScreen />;
  }

  return (
    <NavigationContainer theme={MyTheme}>
      <Stack.Navigator 
        screenOptions={{ 
          headerShown: false,
          animation: 'slide_from_right'
        }}
      >
        <Stack.Screen name="Main" component={MainTabs} />
        <Stack.Screen name="Login" component={LoginScreen} options={{ presentation: 'modal' }} />
        <Stack.Screen name="ForgotPassword" component={ForgotPasswordScreen} />
        <Stack.Screen name="Register" component={RegisterScreen} options={{ presentation: 'modal' }} />
        <Stack.Screen name="Product" component={ProductScreen} />
        <Stack.Screen name="Checkout" component={CheckoutScreen} />
        <Stack.Screen name="Payment" component={PaymentScreen} />
        <Stack.Screen name="AddCard" component={AddCardScreen} />
        <Stack.Screen name="Notifications" component={NotificationsScreen} />
        <Stack.Screen name="AllProducts" component={AllProductsScreen} />
        <Stack.Screen name="PrivacyScreen" component={PrivacyScreen} />
        <Stack.Screen name="SupportScreen" component={SupportScreen} />
        <Stack.Screen name="AboutScreen" component={AboutScreen} />
      </Stack.Navigator>
    </NavigationContainer>
  );
}

import React, { createContext, useState, useEffect, useCallback } from 'react';
import AsyncStorage from '@react-native-async-storage/async-storage';

export const CartContext = createContext();

export const CartProvider = ({ children }) => {
  const [cartItems, setCartItems] = useState([]);
  const [savedCards, setSavedCards] = useState([]);
  const [selectedProducts, setSelectedProducts] = useState([]);

  // ===== CARGA INICIAL UNIFICADA =====
  useEffect(() => {
    const loadData = async () => {
      try {
        const [savedCart, storedCards] = await Promise.all([
          AsyncStorage.getItem('cart'),
          AsyncStorage.getItem('savedCards')
        ]);

        // Cargar carrito con validación
        if (savedCart) {
          const parsed = JSON.parse(savedCart);
          setCartItems(Array.isArray(parsed) ? parsed : []);
        }

        // Cargar tarjetas con normalización
        if (storedCards) {
          const parsedCards = JSON.parse(storedCards);
          
          if (Array.isArray(parsedCards)) {
            // Añadir IDs a tarjetas antiguas
            const normalizedCards = parsedCards.map(card => 
              card.id ? card : { 
                ...card, 
                id: `legacy-${Date.now()}-${Math.random().toString(36).slice(2, 9)}`
              }
            );
            setSavedCards(normalizedCards);
          } else {
            setSavedCards([]);
          }
        }
      } catch (error) {
        console.error('Error loading data:', error);
        // Resetear en caso de error grave
        setCartItems([]);
        setSavedCards([]);
      }
    };

    loadData();
  }, []);

  // ===== GUARDADO AUTOMÁTICO =====
  useEffect(() => {
    AsyncStorage.setItem('cart', JSON.stringify(cartItems));
  }, [cartItems]);

  useEffect(() => {
    AsyncStorage.setItem('savedCards', JSON.stringify(savedCards));
  }, [savedCards]);

  // ===== FUNCIONES MEMOIZADAS =====
  const addToCart = useCallback((product) => {
    if (!product?.id || typeof product.price !== 'number') return;

    setCartItems(prev => {
      const existingIndex = prev.findIndex(item => item.id === product.id);
      
      if (existingIndex > -1) {
        const updated = [...prev];
        updated[existingIndex] = {
          ...updated[existingIndex],
          quantity: updated[existingIndex].quantity + 1
        };
        return updated;
      }
      
      return [...prev, { ...product, quantity: 1 }];
    });
  }, []);

  const removeFromCart = useCallback((id) => {
    setCartItems(prev => prev.filter(item => item.id !== id));
  }, []);

  const updateQuantity = useCallback((id, action) => {
    setCartItems(prev => 
      prev.map(item => 
        item.id === id ? {
          ...item,
          quantity: action === 'increase' 
            ? item.quantity + 1 
            : Math.max(1, item.quantity - 1)
        } : item
      )
    );
  }, []);

  const clearCart = useCallback(() => {
    setCartItems([]);
  }, []);

  // REQUIERE parámetro explícito ahora
  const clearCartAfterPurchase = useCallback((purchasedItems) => {
    setCartItems(prev => 
      prev.filter(item => !purchasedItems.some(p => p.id === item.id))
    );
    setSelectedProducts([]);
  }, []);

  const toggleProductSelection = useCallback((product) => {
    setSelectedProducts(prev => 
      prev.some(p => p.id === product.id)
        ? prev.filter(p => p.id !== product.id)
        : [...prev, product]
    );
  }, []);

  const saveCard = useCallback((cardData) => {
    const newCard = {
      ...cardData,
      id: `card-${Date.now()}-${Math.random().toString(36).slice(2, 9)}`
    };
    setSavedCards(prev => [...prev, newCard]);
  }, []);

  const removeCard = useCallback((cardId) => {
    setSavedCards(prev => prev.filter(card => card.id !== cardId));
  }, []);

  return (
    <CartContext.Provider
      value={{
        cartItems,
        savedCards,
        selectedProducts,
        addToCart,
        removeFromCart,
        updateQuantity,
        clearCart,
        clearCartAfterPurchase,
        toggleProductSelection,
        saveCard,
        removeCard
      }}
    >
      {children}
    </CartContext.Provider>
  );
};

FavContext.js

import React, { createContext, useState, useEffect, useCallback, useMemo } from 'react';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { useFocusEffect } from '@react-navigation/native';
import * as Haptics from 'expo-haptics';

export const FavContext = createContext();

export const FavProvider = ({ children }) => {
    const [favorites, setFavorites] = useState([]);
    const [isSyncing, setIsSyncing] = useState(true);

    // ===== SISTEMA DE SINCRONIZACIÓN MEJORADO =====
    useFocusEffect(
        useCallback(() => {
            let isActive = true;
            
            const syncFavorites = async () => {
                try {
                    setIsSyncing(true);
                    const saved = await AsyncStorage.getItem('favorites');
                    if (saved && isActive) {
                        const parsed = JSON.parse(saved);
                        if (Array.isArray(parsed)) {
                            setFavorites(parsed);
                        } else {
                            console.warn('Corrupted favorites data detected. Resetting...');
                            await AsyncStorage.removeItem('favorites');
                        }
                    }
                } catch (error) {
                    console.error('Syncing error:', error);
                } finally {
                    if (isActive) setIsSyncing(false);
                }
            };

            syncFavorites();
            
            return () => { isActive = false; };
        }, [])
    );

    // ===== PERSISTENCIA INTELIGENTE =====
    useEffect(() => {
        if (isSyncing) return;
        
        const saveFavorites = async () => {
            try {
                await AsyncStorage.setItem('favorites', JSON.stringify(favorites));
            } catch (error) {
                console.error('Persist error:', error);
            }
        };
        
        const debounce = setTimeout(saveFavorites, 500);
        return () => clearTimeout(debounce);
    }, [favorites, isSyncing]);

    // ===== TOGGLE CON HÁPTICA Y ANIMACIÓN =====
    const toggleFavorite = useCallback(async (product) => {
        if (!product?.id) return;

        // Feedback háptico
        await Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light);
        
        setFavorites(prev => {
            const exists = prev.some(f => f.id === product.id);
            if (exists) {
                return prev.filter(f => f.id !== product.id);
            } else {
                return [...prev, {...product, favoritedAt: new Date().toISOString()}];
            }
        });
    }, []);

    // ===== FUNCIONES PREMIUM =====
    const clearFavorites = useCallback(() => {
        Haptics.notificationAsync(Haptics.NotificationFeedbackType.Warning);
        setFavorites([]);
    }, []);

    const bulkAddFavorites = useCallback((products) => {
        const newFavorites = products.filter(p => 
            !favorites.some(f => f.id === p.id)
        );
        
        if (newFavorites.length > 0) {
            Haptics.selectionAsync();
            setFavorites(prev => [
                ...prev,
                ...newFavorites.map(p => ({
                    ...p,
                    favoritedAt: new Date().toISOString()
                }))
            ]);
        }
    }, [favorites]);

    const isFavorite = useCallback((productId) => {
        return favorites.some(f => f.id === productId);
    }, [favorites]);

    const getFavoritesByCategory = useCallback((category) => {
        return favorites.filter(f => f.category === category);
    }, [favorites]);

    // ===== OPTIMIZACIÓN DE RENDIMIENTO =====
    const contextValue = useMemo(() => ({
        favorites,
        isSyncing,
        toggleFavorite,
        clearFavorites,
        bulkAddFavorites,
        isFavorite,
        getFavoritesByCategory,
        favoritesCount: favorites.length,
        hasFavorites: favorites.length > 0
    }), [favorites, isSyncing]);

    return (
        <FavContext.Provider value={contextValue}>
            {children}
        </FavContext.Provider>
    );
};

// context/ThemeContext.js
import React, { createContext, useState, useContext } from 'react';
import { themes } from '../constants/theme';

const ThemeContext = createContext();

export const ThemeProvider = ({ children }) => {
  const [currentTheme, setCurrentTheme] = useState('light');
  
  const toggleTheme = () => {
    setCurrentTheme(prevTheme => prevTheme === 'light' ? 'dark' : 'light');
  };
  
  const theme = themes[currentTheme];
  
  return (
    <ThemeContext.Provider value={{ theme, currentTheme, toggleTheme }}>
      {children}
    </ThemeContext.Provider>
  );
};

export const useTheme = () => useContext(ThemeContext);

  const handlePressIn = () => {
    Animated.spring(scaleValue, {
      toValue: 0.95,
      friction: 3,
      useNativeDriver: true,
    }).start();
  };

  const handlePressOut = () => {
    Animated.spring(scaleValue, {
      toValue: 1,
      friction: 6,
      tension: 40,
      useNativeDriver: true,
    }).start();
    
    if (onPress) {
      setTimeout(() => onPress(category), 100);
    }
  };

  return (
    <Pressable
      onPressIn={handlePressIn}
      onPressOut={handlePressOut}
      accessibilityLabel={category.name}

      accessibilityRole="button"

          styles.card,

import React, { useState } from 'react';
import { View, Text, Image, StyleSheet, Pressable, Animated } from 'react-native';
import { Ionicons } from '@expo/vector-icons';

const ProductCard = ({ product, onPress, onFavoritePress }) => {
  const [scaleValue] = useState(new Animated.Value(1));
  const [favoriteScale] = useState(new Animated.Value(1));
  
  if (!product) return null;

  const handlePressIn = () => {
    Animated.spring(scaleValue, {
      toValue: 0.97,
      friction: 3,
      useNativeDriver: true,
    }).start();
  };

  const handlePressOut = () => {
    Animated.spring(scaleValue, {
      toValue: 1,
      friction: 6,
      tension: 40,
      useNativeDriver: true,
    }).start();
  };

  const handleFavoritePress = () => {
    if (onFavoritePress) {
      Animated.sequence([
        Animated.spring(favoriteScale, {
          toValue: 1.3,
          friction: 3,
          useNativeDriver: true,
        }),
        Animated.spring(favoriteScale, {
          toValue: 1,
          friction: 6,
          tension: 40,
          useNativeDriver: true,
        }),
      ]).start();
      
      setTimeout(() => onFavoritePress(product), 200);
    }
  };

  const renderProfessionalBadge = () => {
    return (
      <View style={styles.proBadge}>
        <Text style={styles.proBadgeText}>PRO</Text>
      </View>
    );
  };

  const renderMaterialBadge = () => {
    if (product.material) {
      return (
        <View style={styles.materialBadge}>
          <Text style={styles.materialText}>{product.material}</Text>
        </View>
      );
    }
    return null;
  };

  const renderTags = () => {
    if (product.tags && product.tags.length > 0) {
      return (
        <View style={styles.tagsContainer}>
          {product.tags.map((tag, index) => (
            <View key={index} style={styles.tag}>
              <Text style={styles.tagText}>{tag}</Text>
            </View>
          ))}
        </View>
      );
    }
    return null;
  };

  return (
    <Pressable
      onPress={onPress}
      onPressIn={handlePressIn}
      onPressOut={handlePressOut}
      accessibilityLabel={`Producto profesional: ${product.name}, Precio: $${product.price}`}
      accessibilityRole="button"
    >
      <Animated.View style={[styles.card, { transform: [{ scale: scaleValue }] }]}>
        <View style={styles.imageContainer}>
          <Image 
            source={product.image} 
            style={styles.image} 
            resizeMode="contain"
            accessibilityLabel={`Imagen de ${product.name}`}
          />
          
          {renderProfessionalBadge()}
          
          <Pressable 
            style={styles.favoriteButton}
            onPress={handleFavoritePress}
            accessibilityLabel={product.isFavorite ? "Quitar de favoritos" : "Añadir a favoritos"}
            accessibilityRole="button"
          >
            <Animated.View style={{ transform: [{ scale: favoriteScale }] }}>
              <Ionicons 
                name={product.isFavorite ? "heart" : "heart-outline"} 
                size={24} 
                color={product.isFavorite ? "#d32f2f" : "#333"} 
              />
            </Animated.View>
          </Pressable>
        </View>
        
        <View style={styles.info}>
          <Text 
            style={styles.name} 
            numberOfLines={2}
            accessibilityLabel={`Nombre: ${product.name}`}
          >
            {product.name}
          </Text>
          
          <Text style={styles.category} numberOfLines={1}>
            {product.category}
          </Text>
          
          {renderMaterialBadge()}
          {renderTags()}
          
          <View style={styles.priceContainer}>
            <Text style={styles.price}>${product.price.toFixed(2)}</Text>
          </View>
          
          <View style={styles.shippingBadge}>
            <Ionicons name="rocket" size={14} color="#4CAF50" />
            <Text style={styles.shippingText}>Entrega exprés</Text>
          </View>
        </View>
      </Animated.View>
    </Pressable>
  );
};

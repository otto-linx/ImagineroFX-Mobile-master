import React, { useEffect } from 'react';
import { 
  View, 
  Text, 
  StyleSheet, 
  TouchableOpacity,
  Linking,
  Animated,
  Platform,
  ImageBackground,
  Dimensions
} from 'react-native';
import { Ionicons } from '@expo/vector-icons';
import { LinearGradient } from 'expo-linear-gradient';

const { width } = Dimensions.get('window');

export default function ForgotPasswordScreen({ navigation }) {
  const [bounceValue] = React.useState(new Animated.Value(1));
  const [fadeAnim] = React.useState(new Animated.Value(0));
  const [slideAnim] = React.useState(new Animated.Value(30));

  useEffect(() => {
    // Animación de entrada
    Animated.parallel([
      Animated.timing(fadeAnim, {
        toValue: 1,
        duration: 800,
        useNativeDriver: true,
      }),
      Animated.timing(slideAnim, {
        toValue: 0,
        duration: 600,
        useNativeDriver: true,
      }),
      Animated.loop(
        Animated.sequence([
          Animated.timing(bounceValue, {
            toValue: 1.05,
            duration: 800,
            useNativeDriver: true,
          }),
          Animated.timing(bounceValue, {
            toValue: 1,
            duration: 800,
            useNativeDriver: true,
          }),
        ])
      ).start()
    ]).start();
  }, []);

  const openWhatsApp = () => {
    const phoneNumber = '573137978572';
    const message = 'Hola, necesito ayuda para recuperar mi contraseña.';
    const url = `https://wa.me/${phoneNumber}?text=${encodeURIComponent(message)}`;
    Linking.openURL(url);
  };

  return (
    <ImageBackground 
      source={require('../assets/theater_background.jpg')}
      style={styles.container}
      resizeMode="cover"
    >
      {/* Capa de superposición roja */}
      <LinearGradient
        colors={['rgba(211, 47, 47, 0.9)', 'rgba(178, 8, 8, 0.95)']}
        style={styles.overlay}
      />
      
      {/* Botón de retroceso */}
      <TouchableOpacity 
        style={styles.backButton} 
        onPress={() => navigation.goBack()}
      >
        <Ionicons name="arrow-back" size={24} color="#fff" />
      </TouchableOpacity>

      {/* Contenido principal con animación */}
      <Animated.View style={[
        styles.content,
        { 
          opacity: fadeAnim,
          transform: [{ translateY: slideAnim }] 
        }
      ]}>
        {/* Encabezado */}
        <View style={styles.header}>
          <Text style={styles.title}>Recupera tu Acceso</Text>
          <View style={styles.divider} />
        </View>
        
        {/* Icono animado */}
        <Animated.View style={[
          styles.iconContainer,
          { transform: [{ scale: bounceValue }] }
        ]}>
          <Ionicons name="key" size={40} color="#fff" />
        </Animated.View>

        {/* Descripción */}
        <Text style={styles.description}>
          ¿Problemas para acceder a tu cuenta? Estamos aquí para ayudarte.
          Contáctanos por WhatsApp y nuestro equipo de soporte te guiará.
        </Text>

        {/* Botón WhatsApp */}
        <Animated.View style={{
          transform: [{ scale: bounceValue }]
        }}>
          <TouchableOpacity 
            style={styles.whatsappButton} 
            onPress={openWhatsApp}
            activeOpacity={0.8}
          >
            <Ionicons name="logo-whatsapp" size={28} color="#fff" />
            <View style={styles.buttonTextContainer}>
              <Text style={styles.whatsappText}>Contactar con Soporte</Text>
              <Text style={styles.whatsappSubtext}>Respuesta inmediata</Text>
            </View>
          </TouchableOpacity>
        </Animated.View>

        {/* Información adicional */}
        <View style={styles.infoContainer}>
          <Ionicons name="time-outline" size={18} color="#fff" />
          <Text style={styles.infoText}>Horario de atención: 8am - 6pm</Text>
        </View>
        <View style={styles.infoContainer}>
          <Ionicons name="calendar-outline" size={18} color="#fff" />
          <Text style={styles.infoText}>Lunes a Sábado</Text>
        </View>

        {/* Enlace de regreso */}
        <TouchableOpacity 
          style={styles.linkContainer} 
          onPress={() => navigation.goBack()}
        >
          <Ionicons name="log-in-outline" size={20} color="#ffd54f" />
          <Text style={styles.linkText}>Volver al inicio de sesión</Text>
        </TouchableOpacity>
      </Animated.View>
      
      {/* Pie de página */}
      <View style={styles.footer}>
        <Text style={styles.footerText}>ImagineroFX - Soluciones Escenográficas</Text>
      </View>
    </ImageBackground>
  );
}

const { width } = Dimensions.get('window');
const scale = width / 375;
function normalize(size) {
  const newSize = size * scale;
  return Math.round(PixelRatio.roundToNearestPixel(newSize));
}

// ========== COMPONENTE PRINCIPAL ========== //
export default function HomeScreen({ navigation }) {
  const [search, setSearch] = useState('');
  const [suggestions, setSuggestions] = useState([]);
  const [selectedCategory, setSelectedCategory] = useState(null);
  const [carouselIndex, setCarouselIndex] = useState(0);
  const { cartItems, addToCart } = useContext(CartContext);
  const cartCount = cartItems.reduce((total, item) => total + (item.quantity || 1), 0);

  // Agrega esta constante:
  const carouselImages = [
    require('../assets/banner1.jpg'),
    require('../assets/banner2.jpg'),
    require('../assets/banner3.jpg'),
  ];

  // Carrusel automático
  useEffect(() => {
    const interval = setInterval(() => {
      setCarouselIndex(prev => (prev === carouselImages.length - 1 ? 0 : prev + 1));
    }, 5000);
    return () => clearInterval(interval);
  }, [carouselImages.length]);
  
  const filteredProducts = products.filter(
    (p) => {
      const categoryMatch = !selectedCategory || 
        p.category === selectedCategory.value;
      const searchMatch = p.name.toLowerCase().includes(search.toLowerCase());
      return categoryMatch && searchMatch;
    }
  );

  const onChangeSearch = (text) => {
    setSearch(text);
    if (text.length > 0) {
      const matches = products.filter((p) =>
        p.name.toLowerCase().includes(text.toLowerCase())
      );
      setSuggestions(matches.slice(0, 5));
    } else {
      setSuggestions([]);
    }
  };

  const toggleCategory = (category) => {
    setSelectedCategory(prev => 
      prev?.id === category.id ? null : category
    );
  };

  const handleAddToCart = (product) => {
    addToCart({...product, quantity: 1});
    Alert.alert('Añadido', `${product.name} agregado al carrito`);
  };

  const handleProductPress = (product) => {
    navigation.navigate('Product', { product });
  };

  return (
    <View style={styles.container}>
      {/* Header */}
      <View style={styles.headerContainer}>
        <View style={styles.headerBackground}>
          <View style={styles.topBar}>
            <View style={styles.logoContainer}>
              <Text style={styles.logo}>ImagineroFX</Text>
            </View>
            <View style={styles.iconsContainer}>
              <TouchableOpacity 
                style={styles.cartButton} 
                onPress={() => navigation.navigate('Cart')}
                accessibilityLabel="Carrito de compras"
                accessibilityHint={`Ver carrito con ${cartCount} productos`}
                accessibilityRole="button"
              >
                <Ionicons name="cart-outline" size={normalize(22)} color="#d32f2f" />
                {cartCount > 0 && (
                  <View style={styles.badge}>
                    <Text style={styles.badgeText}>{cartCount}</Text>
                  </View>
                )}
              </TouchableOpacity>
            </View>
          </View>
          <View style={styles.searchRow}>
            <Ionicons name="search" size={normalize(20)} color="#d32f2f" style={styles.searchIcon} />
            <TextInput
              placeholder="Buscar productos..."
              placeholderTextColor="#d32f2f"
              value={search}
              onChangeText={onChangeSearch}
              style={styles.searchInput}
              accessibilityLabel="Buscar productos"
            />
          </View>
        </View>
        {suggestions.length > 0 && (
          <View style={styles.suggestionsBar}>
            {suggestions.map((item) => (
              <TouchableOpacity
                key={item.id}
                style={styles.suggestionItem}
                onPress={() => {
                  setSearch(item.name);
                  setSuggestions([]);
                  navigation.navigate('Product', { product: item });
                }}
                accessibilityLabel={`Sugerencia: ${item.name}`}
                accessibilityRole="button"
              >
                <Image 
                  source={item.image} 
                  style={styles.suggestionImage} 
                  resizeMode="cover"
                />
                <View style={styles.suggestionTextContainer}>
                  <Text 
                    style={styles.suggestionText} 
                    numberOfLines={1}
                  >
                    {item.name}
                  </Text>
                  <Text 
                    style={styles.suggestionPrice}
                  >
                    ${item.price.toFixed(2)}
                  </Text>
                </View>
              </TouchableOpacity>
            ))}
          </View>
        )}
      </View>

      <ScrollView style={styles.scrollView}>

          <FlatList
            data={[

            horizontal
            pagingEnabled

            renderItem={({ item }) => (
              <Image
                source={item}
                style={styles.carouselImage}
                resizeMode="cover"
              />
            )}
            onViewableItemsChanged={(info) => 
              setCarouselIndex(info.viewableItems[0].index)
            }
          />
          <View style={styles.carouselDots}>
            {[0, 1, 2].map((_, index) => (
              <View
                key={index}
                style={[
                  styles.carouselDot,
                  carouselIndex === index && styles.activeDot
                ]}
              />
            ))}
          </View>
        </View>

        {/* Sección Inspiradora */}
        <View style={styles.inspirationSection}>
          <Image
            source={require('../assets/inspiration.jpg')}

            resizeMode="cover"
          />
          <View style={styles.inspirationTextContainer}>

              En cada escenario, los props y la escenografía no son solo adornos, sino huellas de un mundo creado para emocionar, transportar y dar vida a historias que quedan en la memoria del espectador.

            <Text style={styles.floatingBadgeText}>{cartCount}</Text>
          </View>
        )}
      </TouchableOpacity>
    </View>
  );
}

import React, { useState, useEffect } from 'react';
import {
  View,
  Text,
  TextInput,
  TouchableOpacity,
  StyleSheet,
  Animated,
  KeyboardAvoidingView,
  Platform,
  ActivityIndicator,
  ScrollView,
} from 'react-native';
import { Ionicons } from '@expo/vector-icons';
import { supabase } from '../supabaseClient';
import { LinearGradient } from 'expo-linear-gradient';
import Toast from 'react-native-toast-message'; // Asegúrate de tener esto

export default function LoginScreen({ navigation, route }) {
  const { fromPayment } = route?.params || {}; // Nuevo: recibir parámetro de redirección
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [loading, setLoading] = useState(false);
  const [showPassword, setShowPassword] = useState(false);
  const fadeAnim = React.useRef(new Animated.Value(0)).current;

  // Mostrar mensaje si viene desde PaymentScreen
  useEffect(() => {
    if (fromPayment) {
      Toast.show({
        type: 'info',
        text1: 'Necesitas iniciar sesión',
        text2: 'Primero debes iniciar sesión o registrarte para finalizar tu compra',
        visibilityTime: 3000,
      });
    }
  }, [fromPayment]);

  useEffect(() => {
    Animated.timing(fadeAnim, {
      toValue: 1,
      duration: 700,
      useNativeDriver: true,
    }).start();
  }, []);

  const handleLogin = async () => {
    if (!email || !password) {
      Toast.show({
        type: 'error',
        text1: 'Campos requeridos',
        text2: 'Por favor completa todos los campos.',
      });
      return;
    }

    setLoading(true);

    try {
      const { error } = await supabase.auth.signInWithPassword({ email, password });

      if (error) {
        Toast.show({
          type: 'error',
          text1: 'Error de acceso',
          text2: error.message,
        });
      } else {
        Toast.show({
          type: 'success',
          text1: 'Bienvenido',
          text2: 'Inicio de sesión exitoso.',
          visibilityTime: 2000,
          onHide: () => {
            // Si viene de PaymentScreen, regresa atrás, si no, va al Home
            if (fromPayment) {
              navigation.goBack();
            } else {
              navigation.reset({
                index: 0,
                routes: [{ name: 'Main', params: { screen: 'Home' } }]
              });
            }
          },
        });
      }
    } catch (err) {
      Toast.show({
        type: 'error',
        text1: 'Error inesperado',
        text2: err.message,
      });
    } finally {
      setLoading(false);
    }
  };

  return (
    <KeyboardAvoidingView
      behavior={Platform.OS === 'ios' ? 'padding' : 'height'}
      style={styles.container}
    >
      <LinearGradient colors={['#f8f9fa', '#e0e0e0']} style={styles.background}>
        <ScrollView contentContainerStyle={styles.scrollContent}>
          {/* Botón de retroceso */}
          <TouchableOpacity style={styles.backButton} onPress={() => navigation.goBack()}>
            <Ionicons name="arrow-back" size={24} color="#333" />
          </TouchableOpacity>

          {/* Formulario de login */}
          <Animated.View style={[styles.formContainer, { opacity: fadeAnim }]}>
            <Text style={styles.title}>Bienvenido de vuelta</Text>
            <Text style={styles.subtitle}>Inicia sesión para continuar</Text>

            {/* Email */}
            <View style={styles.inputContainer}>
              <Ionicons name="mail-outline" size={20} color="#999" style={styles.icon} />
              <TextInput
                placeholder="Correo electrónico"
                value={email}
                onChangeText={setEmail}
                style={styles.input}
                keyboardType="email-address"
                autoCapitalize="none"
                placeholderTextColor="#999"
              />
            </View>

            {/* Contraseña */}
            <View style={styles.inputContainer}>
              <Ionicons name="lock-closed-outline" size={20} color="#999" style={styles.icon} />
              <TextInput
                placeholder="Contraseña"
                value={password}
                onChangeText={setPassword}
                style={styles.input}
                secureTextEntry={!showPassword}
                placeholderTextColor="#999"
              />
              <TouchableOpacity onPress={() => setShowPassword(!showPassword)}>
                <Ionicons
                  name={showPassword ? 'eye-off-outline' : 'eye-outline'}
                  size={20}
                  color="#999"
                />
              </TouchableOpacity>
            </View>

            {/* ¿Olvidaste tu contraseña? */}
            <TouchableOpacity
              onPress={() => navigation.navigate('ForgotPassword')}
            >
              <Text style={styles.forgotPassword}>¿Olvidaste tu contraseña?</Text>
            </TouchableOpacity>

            {/* Botón login */}
            <TouchableOpacity style={styles.loginButton} onPress={handleLogin} disabled={loading}>
              {loading ? (
                <ActivityIndicator color="#fff" />
              ) : (
                <Text style={styles.loginButtonText}>Iniciar Sesión</Text>
              )}
            </TouchableOpacity>

}

  const handleInputChange = (name, value) => {
    setFormData({
      ...formData,
      [name]: value
    });
  };

  const validateForm = () => {
    if (!formData.nombre.trim()) {
      Alert.alert('Campo requerido', 'Por favor ingresa tu nombre completo');
      return false;
    }
    if (!formData.telefono.trim() || formData.telefono.length < 7) {
      Alert.alert('Teléfono inválido', 'Por favor ingresa un número de teléfono válido');
      return false;
    }
    if (deliveryOption === 'domicilio' && !formData.direccion.trim()) {
      Alert.alert('Dirección requerida', 'Por favor ingresa tu dirección para la entrega');
      return false;
    }
    if (deliveryOption === 'domicilio' && !formData.ciudad.trim()) {
      Alert.alert('Ciudad requerida', 'Por favor ingresa tu ciudad');
      return false;
    }
    return true;
  };

  const sendWhatsAppMessage = () => {
    if (!validateForm()) return;

    const phoneNumber = '573137978572';
    
    // Construye el mensaje sin símbolos que causen problemas en el UI
    let message = `Hola ImagineroFX, soy ${formData.nombre}. Quiero hacer un pedido contra entrega.\n\n`;
    message += `Productos solicitados:\n`;

    selectedProducts.forEach(item => {
      message += `- ${item.name} x${item.quantity}: ${(item.price * item.quantity).toLocaleString('es-CO', { 
        style: 'currency', 
        currency: 'COP', 
        minimumFractionDigits: 0 
      })}\n`;
    });

    message += `\nTotal: ${formattedTotal}\n`;
    message += `Datos de contacto:\n`;
    message += `Teléfono: ${formData.telefono}\n`;

    if (deliveryOption === 'domicilio') {
      message += `Dirección: ${formData.direccion}, ${formData.ciudad}\n`;
    } else {
      message += `Recogeré en tienda\n`;
    }

    if (formData.notas) {
      message += `\nNotas adicionales:\n${formData.notas}`;
    }

    message += `\nPor favor confirmen mi pedido. ¡Gracias!`;

    const encodedMessage = encodeURIComponent(message);
    const url = `https://wa.me/${phoneNumber}?text=${encodedMessage}`;

    Linking.openURL(url).catch(() => {
      Alert.alert('Error', 'No se pudo abrir WhatsApp. Asegúrate de tener la aplicación instalada.');
    });
  };

  return (
    <View style={styles.container}>
      {/* Encabezado */}
      <ImageBackground 
        source={require('../assets/red_curtain_pattern.jpg')}
        style={styles.header}
        imageStyle={{ borderBottomLeftRadius: 25, borderBottomRightRadius: 25 }}
      >
        <View style={styles.headerOverlay} />
        <Text style={styles.headerTitle}>Finalización de Pedido</Text>
        <Text style={styles.headerSubtitle}>Pago contra entrega</Text>
      </ImageBackground>

      <ScrollView contentContainerStyle={styles.content}>
        {/* Sección de Información de Contacto */}
        <View style={styles.section}>
          <Text style={styles.sectionTitle}>Información de Contacto</Text>
          <View style={styles.inputContainer}>
            <MaterialCommunityIcons name="account" size={24} color="#d32f2f" style={styles.inputIcon} /> 
            <TextInput
              style={styles.input}
              placeholder="Nombre completo"
              placeholderTextColor="#999"
              value={formData.nombre}
              onChangeText={(text) => handleInputChange('nombre', text)}
            />
          </View>
          <View style={styles.inputContainer}>
            <MaterialCommunityIcons name="phone" size={24} color="#d32f2f" style={styles.inputIcon} />
            <TextInput
              style={styles.input}
              placeholder="Teléfono"
              placeholderTextColor="#999"
              keyboardType="phone-pad"
              value={formData.telefono}
              onChangeText={(text) => handleInputChange('telefono', text)}
            />
          </View>
        </View>

        {/* Sección de Método de Entrega */}
        <View style={styles.section}>
          <Text style={styles.sectionTitle}>Método de Entrega</Text>
          <TouchableOpacity 
            style={[styles.deliveryOption, deliveryOption === 'domicilio' && styles.selectedOption]}
            onPress={() => setDeliveryOption('domicilio')}
          >
            <MaterialCommunityIcons name="truck-delivery" size={32} color={deliveryOption === 'domicilio' ? '#d32f2f' : '#666'} />
            <View style={styles.optionTextContainer}>
              <Text style={styles.optionTitle}>Entrega a domicilio</Text>
              <Text style={styles.optionDescription}>Recibe tus productos en la comodidad de tu hogar o lugar de trabajo</Text>
            </View>
            {deliveryOption === 'domicilio' && <Ionicons name="checkmark-circle" size={24} color="#4CAF50" />}
          </TouchableOpacity>

          <TouchableOpacity 
            style={[styles.deliveryOption, deliveryOption === 'tienda' && styles.selectedOption]}
            onPress={() => setDeliveryOption('tienda')}
          >
            <MaterialCommunityIcons name="store" size={32} color={deliveryOption === 'tienda' ? '#d32f2f' : '#666'} />
            <View style={styles.optionTextContainer}>
              <Text style={styles.optionTitle}>Recoger en tienda</Text>
              <Text style={styles.optionDescription}>Recoge tus productos en nuestro almacén principal</Text>
            </View>
            {deliveryOption === 'tienda' && <Ionicons name="checkmark-circle" size={24} color="#4CAF50" />}
          </TouchableOpacity>

          {/* Campos de entrega a domicilio */}
          {deliveryOption === 'domicilio' && (
            <>
              <View style={styles.inputContainer}>
                <MaterialCommunityIcons name="home-map-marker" size={24} color="#d32f2f" style={styles.inputIcon} />
                <TextInput
                  style={styles.input}
                  placeholder="Dirección completa"
                  placeholderTextColor="#999"
                  value={formData.direccion}
                  onChangeText={(text) => handleInputChange('direccion', text)}
                />
              </View>
              <View style={styles.inputContainer}>
                <MaterialCommunityIcons name="city" size={24} color="#d32f2f" style={styles.inputIcon} />
                <TextInput
                  style={styles.input}
                  placeholder="Ciudad"
                  placeholderTextColor="#999"
                  value={formData.ciudad}
                  onChangeText={(text) => handleInputChange('ciudad', text)}
                />
              </View>
            </>
          )}
        </View>

        {/* Detalles Adicionales */}
        <View style={styles.section}>
          <Text style={styles.sectionTitle}>Detalles Adicionales</Text>
          <View style={styles.inputContainer}>
            <MaterialCommunityIcons name="note-text" size={24} color="#d32f2f" style={styles.inputIcon} />
            <TextInput
              style={[styles.input, { height: 80 }]}
              placeholder="Notas adicionales (instrucciones de entrega, horarios preferidos, etc.)"
              placeholderTextColor="#999"
              multiline
              value={formData.notas}
              onChangeText={(text) => handleInputChange('notas', text)}
            />
          </View>
        </View>

        {/* Resumen del Pedido */}
        <View style={styles.section}>
          <Text style={styles.sectionTitle}>Resumen de Pedido</Text>
          <View style={styles.productsContainer}>
            {selectedProducts.map((product, index) => (
              <View key={index} style={styles.productItem}>
                <Text style={styles.productName}>{product.name} x{product.quantity}</Text>
                <Text style={styles.productPrice}>
                  {(product.price * product.quantity).toLocaleString('es-CO', { 
                    style: 'currency', 
                    currency: 'COP', 
                    minimumFractionDigits: 0 
                  })}
                </Text>
              </View>
            ))}
          </View>
          <View style={styles.totalContainer}>
            <Text style={styles.totalLabel}>Total a pagar:</Text>
            <Text style={styles.totalAmount}>{formattedTotal}</Text>
          </View>
        </View>
      </ScrollView>

      {/* Footer */}
      <View style={styles.footer}>
        <TouchableOpacity 
          style={styles.whatsappButton}
          onPress={sendWhatsAppMessage}
        >
          <Ionicons name="logo-whatsapp" size={28} color="#fff" />
          <Text style={styles.whatsappText}>Enviar pedido por WhatsApp</Text>
        </TouchableOpacity>
        <TouchableOpacity 
          style={styles.backButton}
          onPress={() => navigation.goBack()}
        >
          <Text style={styles.backText}>Volver al carrito</Text>
        </TouchableOpacity>
      </View>
    </View>
  );
}

function getSaludo() {

  return 'Buenas noches';

import React, { useEffect, useState, useMemo } from 'react';
import {
  View,
  Text,
  StyleSheet,
  ActivityIndicator,
  Image,
  TouchableOpacity,
  ScrollView,
  Platform,
  Animated
} from 'react-native';
import { Ionicons } from '@expo/vector-icons';
import { supabase } from '../supabaseClient';
import { useFocusEffect } from '@react-navigation/native';

// Paleta de colores rojos
const redTheme = {
  primary: '#FF3B30',
  secondary: '#FF6B6B',
  background: '#FFF5F5',
  card: '#FFFFFF',
  text: '#1A1A1A',
  textSecondary: '#5A5A5A',
  divider: '#E0E0E0',
  error: '#E74C3C',
  neutral: '#F8F9FA'
};

export default function AccountScreen({ navigation }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const [fadeAnim] = useState(new Animated.Value(0));
  
  const styles = useMemo(() => createStyles(redTheme), [redTheme]);

  useFocusEffect(
    React.useCallback(() => {
      Animated.timing(fadeAnim, {
        toValue: 1,
        duration: 500,
        useNativeDriver: true
      }).start();
      
      fetchUser();
    }, [])
  );

  const fetchUser = async () => {
    try {
      const { data } = await supabase.auth.getUser();
      setUser(data?.user || null);
    } catch (error) {
      console.error('Error al cargar usuario:', error);
    } finally {
      setLoading(false);
    }
  };

  const handleLogout = async () => {
    await supabase.auth.signOut();
    setUser(null);
    navigation.navigate('Home');
  };

  const formatDate = (dateString) => {
    if (!dateString) return 'N/A';
    const options = { year: 'numeric', month: 'long', day: 'numeric' };
    return new Date(dateString).toLocaleDateString(undefined, options);
  };

  if (loading) {
    return (
      <View style={styles.loadingContainer}>
        <ActivityIndicator size="large" color={redTheme.primary} />
        <Text style={styles.loadingText}>Cargando tu cuenta...</Text>
      </View>
    );
  }

  return (
    <Animated.View style={{ flex: 1, opacity: fadeAnim }}>
      <ScrollView contentContainerStyle={{ paddingBottom: 40 }}>
        {/* Header */}
        <View style={styles.header}>
          <TouchableOpacity 
            style={styles.backButton}
            onPress={() => navigation.goBack()}
          >
            <Ionicons name="arrow-back" size={24} color={redTheme.icon} />
          </TouchableOpacity>
          <Text style={styles.title}>Mi Cuenta</Text>
        </View>

        {/* Perfil del usuario con efecto gradiente */}
        <View style={styles.profileSection}>
          <View style={styles.avatarContainer}>
            <View style={styles.avatarBorder}>
              <Image
                source={{
                  uri: user
                    ? `https://ui-avatars.com/api/?name=${encodeURIComponent(user.email.split('@')[0])}&background=${redTheme.primary.replace('#', '')}&color=fff` 
                    : `https://ui-avatars.com/api/?name=Invitado&background=${redTheme.neutral.replace('#', '')}&color=000000`, 
                }}
                style={styles.avatar}
              />
            </View>
          </View>
          
          {user ? (
            <>
              <Text style={styles.userName}>{user.email.split('@')[0]}</Text>
              <Text style={styles.userEmail}>{user.email}</Text>
              
              {/* Badge de premium */}
              <View style={styles.premiumBadge}>
                <Ionicons name="star" size={16} color="#fff" />
                <Text style={styles.premiumText}>Cuenta Premium</Text>
              </View>

              {/* Información detallada */}
              <View style={styles.userInfoContainer}>
                <View style={styles.infoItem}>
                  <Ionicons name="calendar-outline" size={18} color={redTheme.textSecondary} />
                  <Text style={styles.infoText}>
                    Miembro desde: {formatDate(user.created_at)}
                  </Text>
                </View>
                <View style={styles.infoItem}>
                  <Ionicons name="key-outline" size={18} color={redTheme.textSecondary} />
                  <Text style={styles.infoText}>
                    Último acceso: {formatDate(new Date())}
                  </Text>
                </View>
              </View>
            </>
          ) : (
            <Text style={styles.guestText}>Invitado</Text>
          )}
        </View>

        {/* Estadísticas mejoradas */}
        {user && (
          <View style={styles.statsContainer}>
            <View style={styles.statItem}>
              <Text style={styles.statNumber}>24</Text>
              <Text style={styles.statLabel}>Actividades</Text>
            </View>
            <View style={styles.statItem}>
              <Text style={styles.statNumber}>18</Text>
              <Text style={styles.statLabel}>Completadas</Text>
            </View>
            <View style={styles.statItem}>
              <Text style={styles.statNumber}>92%</Text>
              <Text style={styles.statLabel}>Éxito</Text>
            </View>
          </View>
        )}

        {/* Sección de opciones actualizada */}
        <View style={styles.optionsSection}>
          <TouchableOpacity 
            style={styles.optionItem}
            onPress={() => navigation.navigate('PrivacyScreen')}
          >
            <View style={styles.optionIcon}>
              <Ionicons name="shield-checkmark-outline" size={22} color={redTheme.icon} />
            </View>
            <Text style={styles.optionText}>Privacidad y Seguridad</Text>
            <Ionicons name="chevron-forward" size={18} color={redTheme.textSecondary} />
          </TouchableOpacity>
          
          <TouchableOpacity 
            style={styles.optionItem}
            onPress={() => navigation.navigate('SupportScreen')}
          >
            <View style={styles.optionIcon}>
              <Ionicons name="help-circle-outline" size={22} color={redTheme.icon} />
            </View>
            <Text style={styles.optionText}>Ayuda y Soporte</Text>
            <Ionicons name="chevron-forward" size={18} color={redTheme.textSecondary} />
          </TouchableOpacity>
          
          <TouchableOpacity 
            style={styles.optionItem}
            onPress={() => navigation.navigate('AboutScreen')}
          >
            <View style={styles.optionIcon}>
              <Ionicons name="information-circle-outline" size={22} color={redTheme.icon} />
            </View>
            <Text style={styles.optionText}>Acerca de la App</Text>
            <Ionicons name="chevron-forward" size={18} color={redTheme.textSecondary} />
          </TouchableOpacity>
        </View>

        {/* Botones principales */}
        <View style={styles.buttonsContainer}>
          {user ? (
            <TouchableOpacity
              style={styles.logoutButton}
              onPress={handleLogout}
            >
              <Ionicons name="log-out-outline" size={20} color={redTheme.error} style={styles.buttonIcon} />
              <Text style={styles.logoutButtonText}>Cerrar Sesión</Text>
            </TouchableOpacity>
          ) : (
            <>
              <TouchableOpacity
                style={[styles.loginButton, { backgroundColor: redTheme.primary }]}
                onPress={() => navigation.navigate('Login')}
              >
                <Ionicons name="log-in-outline" size={20} color="#fff" style={styles.buttonIcon} />
                <Text style={styles.loginButtonText}>Iniciar Sesión</Text>
              </TouchableOpacity>
              <TouchableOpacity
                style={[styles.signupButton, { borderColor: redTheme.primary }]}
                onPress={() => navigation.navigate('Register')}
              >
                <Ionicons name="person-add-outline" size={20} color={redTheme.primary} style={styles.buttonIcon} />
                <Text style={[styles.signupButtonText, { color: redTheme.primary }]}>Crear Cuenta</Text>
              </TouchableOpacity>
            </>
          )}
        </View>

        {/* Footer mejorado */}
        <View style={styles.footer}>
          <Text style={styles.footerText}>
            Versión 2.1.0 • Última actualización: {formatDate(new Date())}
          </Text>
          <View style={styles.badgeContainer}>
            <View style={styles.badge}>
              <Text style={styles.badgeText}>PRO</Text>
            </View>
          </View>
        </View>
      </ScrollView>
    </Animated.View>
  );
}

export default function AllProductsScreen({ navigation }) {
  const [search, setSearch] = useState('');
  const [selectedCategory, setSelectedCategory] = useState('all');
  const [showFilters, setShowFilters] = useState(false);
  const [selectedMaterial, setSelectedMaterial] = useState('');
  const [selectedColor, setSelectedColor] = useState('');
  const [priceRange, setPriceRange] = useState([0, 5000]);
  const [filteredProducts, setFilteredProducts] = useState(products);
  const [materialFilter, setMaterialFilter] = useState('');
  const [colorFilter, setColorFilter] = useState('');
  const [priceFilter, setPriceFilter] = useState([0, 5000]);

  const allMaterials = [...new Set(products.map(p => p.material))];
  const allColors = [...new Set(products.flatMap(p => (p.colors ?? '').split(', ')))];

  const handleSearch = (text) => {
    setSearch(text);
    filterProducts(text, selectedCategory, materialFilter, colorFilter, priceFilter);
  };

  const handleCategory = (categoryId) => {
    setSelectedCategory(categoryId);
    filterProducts(search, categoryId, materialFilter, colorFilter, priceFilter);
  };

  const applyFilters = () => {
    filterProducts(search, selectedCategory, materialFilter, colorFilter, priceFilter);
    setShowFilters(false);
  };

  const resetFilters = () => {
    setSearch('');
    setSelectedCategory('all');
    setMaterialFilter('');
    setColorFilter('');
    setPriceFilter([0, 5000]);
    setFilteredProducts(products);
    setShowFilters(false);
  };

  const filterProducts = (searchText = '', categoryId = 'all', material = '', color = '', price = [0, 5000]) => {
    let result = [...products];
    
    // Filtrar por categoría
    if (categoryId !== 'all') {
      result = result.filter(p => p.category.toLowerCase().includes(categoryId.toLowerCase()));
    }
    
    // Filtrar por búsqueda
    if (searchText) {
      const searchLower = searchText.toLowerCase();
      result = result.filter(p => 
        p.name.toLowerCase().includes(searchLower) || 
        p.category.toLowerCase().includes(searchLower)
      );
    }
    
    // Filtrar por material
    if (material) {
      result = result.filter(p => p.material.toLowerCase().includes(material.toLowerCase()));
    }
    
    // Filtrar por color
    if (color) {
      result = result.filter(p => p.colors.toLowerCase().includes(color.toLowerCase()));
    }
    
    // Filtrar por precio
    result = result.filter(p => {
      const productPrice = parseFloat(p.price);
      return productPrice >= price[0] && productPrice <= price[1];
    });
    
    setFilteredProducts(result);
  };

  const renderProduct = ({ item }) => (
    <TouchableOpacity 
      style={styles.productCard}
      onPress={() => navigation.navigate('Product', { product: item })}
      activeOpacity={0.9}
    >
      <View style={styles.imageContainer}>

          <Text style={styles.productPrice}>

          </Text>
        </View>
      </View>
    </TouchableOpacity>
  );

  const renderCategory = ({ item }) => {
    const isSelected = selectedCategory === item.id;
    return (
      <TouchableOpacity
        style={[
          styles.categoryButton,
          isSelected ? styles.categorySelected : styles.categoryUnselected
        ]}
        onPress={() => handleCategory(item.id)}
        activeOpacity={0.85}
      >
        <View style={[
          styles.categoryIconContainer,
          isSelected && styles.categoryIconSelected
        ]}>
          <Ionicons

          />
        </View>
        <Text style={isSelected ? styles.categoryTextSelected : styles.categoryTextUnselected}>
          {item.name}
        </Text>
      </TouchableOpacity>
    );
  };

  return (
    <View style={styles.container}>
      {/* Botón de retroceso */}
      <TouchableOpacity style={styles.backButton} onPress={() => navigation.goBack()}>

          onChangeText={handleSearch}
          style={styles.searchInput}
        />
      </View>
      
      {/* Categorías */}

        >
          {categories.map(category => (

              {renderCategory({ item: category })}

          ))}
        </ScrollView>

CheckoutScreen.js

import React, { useContext, useState } from 'react';
import {
  View,
  Text,
  TouchableOpacity,
  StyleSheet,
  ScrollView,
  Image,
  Alert,
  Animated,
  Dimensions
} from 'react-native';
import { MaterialCommunityIcons } from '@expo/vector-icons';
import { CartContext } from '../context/CartContext';

const { width } = Dimensions.get('window');

export default function CheckoutScreen({ navigation }) {
  const { cartItems } = useContext(CartContext);
  const [selectedProducts, setSelectedProducts] = useState([]);
  const [scaleAnim] = useState(new Animated.Value(1));

  const toggleProductSelection = (product) => {
    Animated.sequence([
      Animated.timing(scaleAnim, {
        toValue: 0.95,
        duration: 100,
        useNativeDriver: true
      }),
      Animated.timing(scaleAnim, {
        toValue: 1,
        duration: 200,
        useNativeDriver: true
      })
    ]).start();
    
    setSelectedProducts((prev) =>
      prev.some((p) => p.id === product.id)
        ? prev.filter((p) => p.id !== product.id)
        : [...prev, product]
    );
  };

  const handleProceedToPayment = () => {
    if (selectedProducts.length === 0) {
      Alert.alert('Selección requerida', 'Por favor selecciona al menos un producto para continuar.');
      return;
    }
    navigation.navigate('Payment', { selectedProducts });
  };

  const total = selectedProducts.reduce(
    (sum, item) => sum + item.price * (item.quantity || 1),
    0
  );

  return (
    <View style={styles.container}>
      {/* Encabezado con efecto de telón */}
      <View style={styles.header}>
        <View style={styles.headerOverlay} />
        <Text style={styles.title}>Finalización de Compra</Text>
        <Text style={styles.subtitle}>Selecciona los productos a pagar</Text>
      </View>

      <ScrollView contentContainerStyle={styles.contentContainer}>
        {cartItems.length === 0 ? (
          <View style={styles.emptyContainer}>
            <MaterialCommunityIcons name="theater" size={60} color="#d32f2f" />
            <Text style={styles.emptyTitle}>Carrito Vacío</Text>
            <Text style={styles.emptyText}>No hay productos en tu carrito</Text>
            <TouchableOpacity 
              style={styles.shoppingButton}
              onPress={() => navigation.goBack()}
            >
              <Text style={styles.shoppingText}>Volver a productos</Text>
            </TouchableOpacity>
          </View>
        ) : (
          <>
            <Text style={styles.sectionTitle}>Tus Productos Escenográficos</Text>
            
            {cartItems.map((product) => {
              const isSelected = selectedProducts.some((p) => p.id === product.id);
              return (
                <Animated.View
                  key={product.id}
                  style={[
                    styles.productContainer,
                    isSelected && styles.selectedProduct,
                    { transform: [{ scale: scaleAnim }] }
                  ]}
                >
                  <TouchableOpacity
                    onPress={() => toggleProductSelection(product)}
                    activeOpacity={0.8}
                    style={styles.productTouchable}
                  >
                    <View style={styles.productContent}>
                      <View style={styles.checkboxContainer}>
                        <View style={[
                          styles.checkbox,
                          isSelected && styles.checkboxSelected
                        ]}>
                          {isSelected && (
                            <MaterialCommunityIcons 
                              name="check" 
                              size={18} 
                              color="#fff" 
                            />
                          )}
                        </View>
                      </View>
                      
                      {product.image ? (
                        <Image
                          source={
                            typeof product.image === 'string'
                              ? { uri: product.image }
                              : product.image
                          }
                          style={styles.productImage}
                        />
                      ) : (
                        <View style={styles.imagePlaceholder}>
                          <MaterialCommunityIcons 
                            name="theater" 
                            size={24} 
                            color="#fff" 
                          />
                        </View>
                      )}
                      
                      <View style={styles.textContainer}>
                        <Text 
                          style={styles.productText}
                          numberOfLines={1}
                        >
                          {product.name || 'Producto escenográfico'}
                        </Text>
                        <Text style={styles.categoryText}>
                          {product.category || 'Escenografía'}
                        </Text>
                      </View>
                      
                      <View style={styles.priceContainer}>
                        <Text style={styles.priceText}>
                          ${Number(product.price).toFixed(2)}
                        </Text>
                        <Text style={styles.quantityText}>
                          x {product.quantity || 1}
                        </Text>
                      </View>
                    </View>
                  </TouchableOpacity>
                </Animated.View>
              );
            })}
          </>
        )}
      </ScrollView>

      {/* Resumen de compra */}
      <View style={styles.summaryContainer}>
        <View style={styles.totalRow}>
          <Text style={styles.totalLabel}>Total seleccionado:</Text>
          <Text style={styles.totalAmount}>
            ${total.toFixed(2)}
          </Text>
        </View>
        
        <TouchableOpacity
          style={[
            styles.payButton,
            selectedProducts.length === 0 && styles.disabledButton,
          ]}
          onPress={handleProceedToPayment}
          disabled={selectedProducts.length === 0}
        >
          <MaterialCommunityIcons 
            name="credit-card-check" 
            size={24} 
            color="#fff" 
          />
          <Text style={styles.payButtonText}>Continuar al pago</Text>
        </TouchableOpacity>
        
        <TouchableOpacity
          style={styles.backButton}
          onPress={() => navigation.goBack()}
        >
          <Text style={styles.backButtonText}>Seguir comprando</Text>
        </TouchableOpacity>
      </View>
    </View>
  );
}
